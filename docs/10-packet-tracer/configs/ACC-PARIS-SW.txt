! ACC-PARIS-SW — Switch 2960-24TT — accès unique du siège
! (postes bureautique, poste bancaire, borne Wi-Fi, téléphone IP)
enable
configure terminal
hostname ACC-PARIS-SW
enable secret Cisco123!
no ip domain-lookup
banner motd #KHS Bank - Acces PARIS - Acces reserve MOM-TECH#

vlan 10
 name BUREAUTIQUE
vlan 20
 name BANCAIRE-KHSCORE
vlan 40
 name WIFI-INVITES
vlan 50
 name VOIP
vlan 99
 name MANAGEMENT
vlan 999
 name NATIVE-INUTILISE
exit

spanning-tree mode rapid-pvst

! --- Deux liaisons montantes redondantes, une vers chaque cœur (RSTP choisit
!     le chemin actif ; si CORE-PARIS-SW1 tombe, ce lien bascule sur SW2). ---
interface FastEthernet0/1
 description LIEN-VERS-CORE-PARIS-SW1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,40,50,99
 switchport trunk native vlan 999
exit

interface FastEthernet0/2
 description LIEN-VERS-CORE-PARIS-SW2
 switchport mode trunk
 switchport trunk allowed vlan 10,20,40,50,99
 switchport trunk native vlan 999
exit

interface FastEthernet0/3
 description PC-BUR-PARIS-1
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
exit

interface FastEthernet0/4
 description PC-BANCAIRE-PARIS-1
 switchport mode access
 switchport access vlan 20
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
exit

interface FastEthernet0/5
 description AP-PARIS
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
exit

! --- Laptop invité en filaire (simplification : module Wi-Fi non compatible dans PT,
!     cf. 01-topologie-et-cablage.md) : même VLAN 40 qu'une connexion sans fil réelle.
!     Sur FastEthernet0/6 (et non 0/7 comme prévu initialement) : câblage réel de la
!     maquette, le câble du laptop et celui du téléphone IP ayant été inversés lors
!     de la construction — sans incidence sur les compétences démontrées. ---
interface FastEthernet0/6
 description LAPTOP-WIFI-PARIS-1
 switchport mode access
 switchport access vlan 40
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 spanning-tree portfast
exit

! --- Port téléphone IP : VLAN voix dédiée + VLAN data pour un PC éventuel derrière le tél.
!     Sur FastEthernet0/7 (et non 0/6) : câblage réel, cf. note ci-dessus. ---
interface FastEthernet0/7
 description IPPHONE-PARIS-1
 switchport mode access
 switchport access vlan 10
 switchport voice vlan 50
 switchport port-security
 switchport port-security maximum 3
 switchport port-security violation restrict
 spanning-tree portfast
exit

interface Vlan99
 ip address 10.10.12.11 255.255.255.192
 no shutdown
exit
ip default-gateway 10.10.12.1
username admmomtech privilege 15 secret Cisco123!
ip domain-name khsbank.internal
crypto key generate rsa modulus 2048
line vty 0 15
 login local
 transport input ssh
exit

end
write memory
