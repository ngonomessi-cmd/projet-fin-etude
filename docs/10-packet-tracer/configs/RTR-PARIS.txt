interface Tunnel0
 description TUNNEL-GRE-VERS-LYON
 ip address 172.16.0.1 255.255.255.252

! --- Routage ---

! --- NAT (Network Address Translation) : seul le trafic réellement destiné à

access-list 100 permit ip 10.10.0.0 0.0.255.255 any
