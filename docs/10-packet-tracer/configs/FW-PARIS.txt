! FW-PARIS — ASA 5505 — pare-feu périmétrique du siège
! Le NAT est géré par RTR-PARIS ; l'ASA assure ici le filtrage et l'inspection stateful.
enable
configure terminal
hostname FW-PARIS

! --- VLAN internes de l'ASA 5505 (le boîtier ne fait pas de trunk sur licence de base) ---
interface Vlan1
 nameif inside
 security-level 100
 ip address 10.10.254.5 255.255.255.248
exit

interface Vlan2
 nameif outside
 security-level 0
 ip address 10.10.254.2 255.255.255.252
exit

! --- Ports physiques rattachés aux VLAN ---
interface Ethernet0/0
 switchport access vlan 2
 no shutdown
exit

interface Ethernet0/1
 description LIEN-VERS-CORE-PARIS-SW1
 switchport access vlan 1
 no shutdown
exit

interface Ethernet0/2
 description LIEN-VERS-CORE-PARIS-SW2
 switchport access vlan 1
 no shutdown
exit

! --- Routage ---
! Par défaut : tout ce qui n'est pas local part vers le routeur (Internet ET site distant,
! RTR-PARIS sait ensuite distinguer via sa route spécifique vers le tunnel).
route outside 0.0.0.0 0.0.0.0 10.10.254.1

! --- Inspection stateful : autoriser explicitement ICMP pour les tests de ping ---
policy-map global_policy
 class inspection_default
  inspect icmp
  inspect http
  inspect ftp
  inspect dns

! --- Comptes et accès de gestion ---
username admmomtech password Cisco123! privilege 15
aaa authentication ssh console LOCAL
crypto key generate rsa modulus 1024
ssh 10.10.0.0 255.255.252.0 inside
ssh timeout 10

end
write memory
